>[!example] Operador %
> Ejecuta el código y descubre la función del operador `%`. Prueba con diferentes valores.
> ```cpp
> #include <iostream>
>int main() {
 >   int paquetesDatos = 14;
 >   int capacidadPaquete = 4;
 >   int sobrante = paquetesDatos % capacidadPaquete;
>
 >   if (sobrante == 0) {
 >       std::cout << "Transmision completa sin restos." << std::endl;
 >   } else {
 >       std::cout << "Alerta: quedan " << sobrante << " datos sueltos." << std::endl;
 >   }
>
 >   return 0;
>}
>```
